// Variable untuk menyimpan statistik summary let stats = { white: { brilliant: 0, great: 0, best: 0, excellent: 0, good: 0, book: 0, inacc: 0, mistake: 0, blunder: 0, accuracy: 0 }, black: { brilliant: 0, great: 0, best: 0, excellent: 0, good: 0, book: 0, inacc: 0, mistake: 0, blunder: 0, accuracy: 0 } }; async function runAnalysis() { if (!selectedPgn) return; // Reset Stats stats = { white: { brilliant: 0, great: 0, best: 0, excellent: 0, good: 0, book: 0, inacc: 0, mistake: 0, blunder: 0, total: 0 }, black: { brilliant: 0, great: 0, best: 0, excellent: 0, good: 0, book: 0, inacc: 0, mistake: 0, blunder: 0, total: 0 } }; document.getElementById('loading').style.display = 'flex'; let temp = new Chess(); temp.load_pgn(selectedPgn); gHistory = temp.history({verbose: true}); temp.reset(); results = []; let lastEval = 0.2; for (let i = 0; i < gHistory.length; i++) { let turn = i % 2 === 0 ? 'white' : 'black'; temp.move(gHistory[i].san); // Simulasikan progress document.getElementById('pct').innerText = Math.round(((i+1)/gHistory.length)*100) + "%"; let curEval = lastEval; // Ambil data eval (Simulasi Engine Logic) try { const r = await fetch(`https://lichess.org/api/cloud-eval?fen=${encodeURIComponent(temp.fen())}`); if (r.ok) { const d = await r.json(); if (d.pvs && d.pvs[0]) { const cp = d.pvs[0].cp; curEval = (temp.turn() === 'b' ? cp : -cp) / 100; } } } catch(e) {} const diff = curEval - lastEval; const absDiff = Math.abs(diff); let r; // --- LOGIKA KLASIFIKASI KETAT (Agar tidak Best Mulu) --- if (i < 8) { r = { sym: '📖', color: 'var(--book)', label: 'BOOK' }; stats[turn].book++; } else if (diff < -3.0) { r = { sym: '??', color: 'var(--blunder)', label: 'BLUNDER' }; stats[turn].blunder++; } else if (diff < -1.0) { r = { sym: '?', color: 'var(--mistake)', label: 'MISTAKE' }; stats[turn].mistake++; } else if (diff < -0.4) { r = { sym: '?!', color: 'var(--inacc)', label: 'INACCURACY' }; stats[turn].inacc++; } else if (diff > 1.5 && lastEval < -0.5) { r = { sym: '!!', color: 'var(--brilliant)', label: 'BRILLIANT' }; stats[turn].brilliant++; } else if (diff > 0.6) { r = { sym: '!', color: 'var(--great)', label: 'GREAT' }; stats[turn].great++; } else if (absDiff <= 0.15) { // Hanya jika sangat presisi r = { sym: '★', color: 'var(--best)', label: 'BEST' }; stats[turn].best++; } else if (absDiff <= 0.3) { r = { sym: '👍', color: 'var(--excellent)', label: 'EXCELLENT' }; stats[turn].excellent++; } else { r = { sym: '✓', color: 'var(--good)', label: 'GOOD' }; stats[turn].good++; } results.push({...r, score: curEval}); lastEval = curEval; } showSummary(); // Tampilkan tabel hasil seperti di gambar kamu game.reset(); gIdx = -1; document.getElementById('loading').style.display = 'none'; draw(); } function showSummary() { // Hitung akurasi kasar const calcAcc = (s) => { let penalty = (s.blunder * 30) + (s.mistake * 15) + (s.inacc * 5); let acc = 100 - (penalty / (results.length/2)); return Math.max(Math.min(acc, 100), 10).toFixed(1); }; // Buat tampilan pop-up Summary (Bisa kamu taruh di dalam UI Card) let summaryHtml = `
Accuracy: ${calcAcc(stats.white)}
Accuracy: ${calcAcc(stats.black)}
${stats.white.brilliant}!! Brilliant${stats.black.brilliant}
${stats.white.best}★ Best${stats.black.best}
${stats.white.blunder}?? Blunder${stats.black.blunder}
`; document.getElementById('list').innerHTML = summaryHtml; }
